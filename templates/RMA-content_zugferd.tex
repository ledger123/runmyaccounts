% PDF/A-3B conformance requires that every stream's /Length key matches the
% actual byte count in the file.  In some pdfTeX versions, file-based streams
% written via \pdfobj stream file{...} receive a /Length derived from the
% uncompressed file size while the stream content is FlateDecode-compressed,
% causing a /Length mismatch that fails VeraPDF validation.
%
% \pdfcompresslevel=0 disables FlateDecode compression for ALL stream objects
% so that pdfLaTeX always writes raw (uncompressed) bytes and the /Length key
% is guaranteed to be exact.  This is set at preamble level so that it is
% already in effect when the pdfx package writes its own /Metadata stream
% inside its \AtBeginDocument hook (which runs before ours).
\pdfcompresslevel=0

% Inject Factur-X XMP as an additional PDF metadata stream.
% This hook runs after the pdfx package registers its own metadata (pdfx
% registers at \usepackage time, we register here in content_zugferd.tex
% which is included later), so our catalog entry is written after pdfx's.
% In pdfTeX, for duplicate catalog keys the last occurrence takes effect.
\AtBeginDocument{%
  % \pdfobjcompresslevel=0 prevents the /Metadata stream from being packed
  % into a compressed object stream (PDF 1.5 object streams), keeping it
  % directly accessible as required by PDF/A-3B.
  \pdfobjcompresslevel=0
  \immediate\pdfobj stream attr{/Type /Metadata /Subtype /XML}%
    file{\jobname-fxmeta.xmp}%
  % Restore object-stream compression for the rest of the document so that
  % other PDF objects benefit from PDF 1.5 object-stream packing.
  \pdfobjcompresslevel=2
  \pdfcatalog{/Metadata \the\pdflastobj\space 0 R}%
}
